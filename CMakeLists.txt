cmake_minimum_required(VERSION 3.24)
project(FaceDetection VERSION 1.0 LANGUAGES CXX)
option(FACE_STATIC_CRT "Use the static MSVC runtime (match OpenCV)" ON)
if(MSVC AND FACE_STATIC_CRT)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(FACE_BUILD_APP "Build the OpenCV webcam application" ON)
option(FACE_SANITIZE "Enable address/undefined sanitizers on GCC/Clang" OFF)
add_library(face_core src/core.cpp)
target_include_directories(face_core PUBLIC src)
if(MSVC)
  target_compile_options(face_core PUBLIC /W4 /permissive- /utf-8)
else()
  target_compile_options(face_core PUBLIC -Wall -Wextra -Wpedantic -Wconversion)
  if(FACE_SANITIZE)
    target_compile_options(face_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(face_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE face_core)
  add_test(NAME core COMMAND core_tests)
endif()
if(FACE_BUILD_APP)
  find_package(OpenCV 4.9 REQUIRED COMPONENTS core imgproc highgui videoio objdetect dnn)
  add_executable(face_detection src/main.cpp src/detector.cpp src/camera.cpp src/render.cpp)
  target_include_directories(face_detection PRIVATE ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(face_detection PRIVATE face_core ${OpenCV_LIBS})
  if(BUILD_TESTING)
    add_test(NAME startup COMMAND ${CMAKE_COMMAND} -DAPP=$<TARGET_FILE:face_detection> -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/startup.cmake)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/models/face_detection_yunet_2023mar.onnx")
      add_test(NAME model_inference COMMAND face_detection --model "${CMAKE_CURRENT_SOURCE_DIR}/models/face_detection_yunet_2023mar.onnx" --check-model)
      set_tests_properties(model_inference PROPERTIES TIMEOUT 30 PASS_REGULAR_EXPRESSION "synthetic-frame inference passed")
    endif()
  endif()
  if(WIN32)
    target_link_libraries(face_detection PRIVATE shell32)
  endif()
endif()
